> Wrangler Cloudflare D1 Database (mystical-data)

# Login / ID
npm install --save-dev wrangler
wrangler login
wrangler whoami

# Create D1 database - ONCE
wrangler d1 create mystical-data

# Apply schema locally
wrangler d1 execute mystical-data --local --file=./schema.sql

# Start local website
wrangler pages dev .

# See local users
wrangler d1 execute mystical-data --local --command="SELECT * FROM users;"

# Delete all local users
wrangler d1 execute mystical-data --local --command="DELETE FROM users;"

# Inspect save data
wrangler d1 execute mystical-data --local --command="SELECT username, save_data FROM users;"

# Database information
wrangler d1 list
wrangler d1 info mystical-data

# Account limits SELECT
npx wrangler d1 execute mystical-data --remote --command "CREATE TABLE IF NOT EXISTS registration_limits (ip TEXT PRIMARY KEY, window_start INTEGER NOT NULL, account_count INTEGER NOT NULL DEFAULT 0);"
